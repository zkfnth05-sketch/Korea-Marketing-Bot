### ① `lib/supabaseBlog.ts` (블로그 DB 조회 유틸리티)
```typescript
import { createClient } from '@supabase/supabase-js';

const supabaseUrl = process.env.NEXT_PUBLIC_SUPABASE_URL || '';
const supabaseAnonKey = process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY || '';

export const supabaseBlog = createClient(supabaseUrl, supabaseAnonKey);

// 프로젝트 환경변수 또는 도메인에 따라 테이블 자동 선택
export const isEasyTax = process.env.NEXT_PUBLIC_SERVICE_ID === 'easytax';
export const BLOG_TABLE = isEasyTax ? 'easytax_blogs' : 'kmarket_blogs';

export interface BlogPost {
  id: number;
  slug: string;
  target_lang: string;
  title: string;
  excerpt: string;
  content_html: string;
  content_md?: string;
  thumbnail_url: string;
  category: string;
  author: string;
  views: number;
  likes: number;
  published_at: string;
  created_at: string;
}

// 1. 특정 언어의 최신 블로그 목록 조회
export async function getBlogPosts(lang: string = 'en', limit: number = 24): Promise<BlogPost[]> {
  const { data, error } = await supabaseBlog
    .from(BLOG_TABLE)
    .select('*')
    .eq('target_lang', lang)
    .order('published_at', { ascending: false })
    .limit(limit);

  if (error) {
    console.error(`[Supabase Blog] Error fetching posts for lang=${lang}:`, error);
    return [];
  }
  return data || [];
}

// 2. 특정 슬러그와 언어의 블로그 상세 조회
export async function getBlogPostBySlug(slug: string, lang: string = 'en'): Promise<BlogPost | null> {
  const { data, error } = await supabaseBlog
    .from(BLOG_TABLE)
    .select('*')
    .eq('slug', slug)
    .eq('target_lang', lang)
    .maybeSingle();

  if (error) {
    console.error(`[Supabase Blog] Error fetching post slug=${slug}, lang=${lang}:`, error);
    return null;
  }
  return data;
}
```

### ③ `app/[lang]/blog/[slug]/page.tsx` (1,500자 상세 칼럼 & SEO 리치 스니펫)
```tsx
import { notFound } from 'next/navigation';
import Link from 'next/link';
import { getBlogPostBySlug, isEasyTax } from '@/lib/supabaseBlog';
import { Metadata } from 'next';

interface Props {
  params: { lang: string; slug: string };
}

const SUPPORTED_LANGUAGES = ['en', 'vi', 'zh', 'ko', 'uz', 'mn', 'ru', 'th', 'id', 'tl', 'ne', 'my', 'ja', 'si', 'km', 'bn', 'ur'];

// 1. Google SEO 리치 스니펫 & hreflang 17개국어 메타데이터 자동 주입
export async function generateMetadata({ params }: Props): Promise<Metadata> {
  const { lang, slug } = params;
  const post = await getBlogPostBySlug(slug, lang);
  if (!post) return { title: 'Article Not Found' };

  const isTax = isEasyTax;
  const baseDomain = isTax ? 'https://ktrs-service.vercel.app' : 'https://ktrs-market.vercel.app';
  const canonicalUrl = `${baseDomain}/${lang}/blog/${slug}`;

  // 17개 언어 hreflang 상호 링크
  const languageAlternates: Record<string, string> = {};
  SUPPORTED_LANGUAGES.forEach((l) => {
    languageAlternates[l] = `${baseDomain}/${l}/blog/${slug}`;
  });

  return {
    title: `${post.title} | ${isTax ? 'EasyTax Korea' : 'K-Market Korea'}`,
    description: post.excerpt,
    alternates: {
      canonical: canonicalUrl,
      languages: languageAlternates,
    },
    openGraph: {
      title: post.title,
      description: post.excerpt,
      url: canonicalUrl,
      type: 'article',
      publishedTime: post.published_at,
      images: post.thumbnail_url ? [{ url: post.thumbnail_url }] : [],
    },
  };
}

export default async function BlogPostDetailPage({ params }: Props) {
  const { lang, slug } = params;
  const post = await getBlogPostBySlug(slug, lang);

  if (!post) {
    notFound();
  }

  const isTax = isEasyTax;
  const baseDomain = isTax ? 'https://ktrs-service.vercel.app' : 'https://ktrs-market.vercel.app';
  const canonicalUrl = `${baseDomain}/${lang}/blog/${slug}`;

  // 2. Schema.org BlogPosting JSON-LD 리치 스니펫 (구글 검색 상단 강조 노출용)
  const jsonLd = {
    '@context': 'https://schema.org',
    '@type': 'BlogPosting',
    headline: post.title,
    description: post.excerpt,
    datePublished: post.published_at,
    author: {
      '@type': 'Organization',
      name: post.author || (isTax ? 'EasyTax Legal Team' : 'K-Market Expat Team'),
      url: baseDomain,
    },
    publisher: {
      '@type': 'Organization',
      name: isTax ? 'EasyTax Korea' : 'K-Market Korea',
      logo: {
        '@type': 'ImageObject',
        url: `${baseDomain}/logo.png`,
      },
    },
    mainEntityOfPage: {
      '@type': 'WebPage',
      '@id': canonicalUrl,
    },
  };

  return (
    <div className="min-h-screen bg-slate-950 text-slate-100 py-12 px-4 sm:px-6 lg:px-8">
      {/* 구글 검색엔진용 JSON-LD 주입 */}
      <script
        type="application/ld+json"
        dangerouslySetInnerHTML={{ __html: JSON.stringify(jsonLd) }}
      />

      <article className="max-w-3xl mx-auto">
        {/* 상단 네비게이션 */}
        <div className="mb-6">
          <Link href={`/${lang}/blog`} className="text-sm font-semibold text-emerald-400 hover:underline">
            ← Back to All Articles
          </Link>
        </div>

        {/* 글 헤더 */}
        <header className="mb-8">
          <span className="text-xs font-semibold text-emerald-400 uppercase tracking-widest bg-emerald-500/10 border border-emerald-500/20 px-3 py-1 rounded-full">
            {post.category || 'Guide'}
          </span>
          <h1 className="text-3xl sm:text-4xl font-extrabold text-white mt-4 leading-tight">
            {post.title}
          </h1>
          <div className="flex items-center text-xs text-slate-500 mt-4 space-x-4">
            <span>By {post.author}</span>
            <span>•</span>
            <span>Published on {new Date(post.published_at).toLocaleDateString()}</span>
          </div>
        </header>

        {/* 대표 썸네일 */}
        {post.thumbnail_url && (
          <div className="mb-8 rounded-2xl overflow-hidden border border-slate-800 bg-slate-900">
            <img src={post.thumbnail_url} alt={post.title} className="w-full h-auto object-cover" />
          </div>
        )}

        {/* 1,500자 본문 HTML 렌더링 */}
        <div 
          className="prose prose-invert prose-emerald max-w-none text-slate-300 leading-relaxed text-base space-y-6"
          dangerouslySetInnerHTML={{ __html: post.content_html }}
        />

        {/* 하단 전환 CTA 박스 (웹 전환 유도) */}
        <div className="mt-12 p-8 bg-slate-900 border border-slate-800 rounded-2xl text-center">
          <h3 className="text-xl font-bold text-white">
            {isTax ? '💰 Check Your 5-Year Expat Tax Refund Free' : '🎁 Find 0 KRW Free Giveaways in Korea'}
          </h3>
          <p className="text-sm text-slate-400 mt-2 max-w-md mx-auto">
            {isTax 
              ? 'Foreign workers can reduce up to 90% of income tax under Korean Tax Law (Article 30). Processed by certified tax agents.'
              : 'Connect with verified expats across South Korea. 100% free instant translation chat in 17 languages.'}
          </p>
          <Link
            href={`/${lang}/welcome?utm_source=blog_cta&utm_medium=organic_article`}
            className="inline-block mt-5 px-6 py-3 bg-emerald-500 hover:bg-emerald-400 text-slate-950 font-bold rounded-xl shadow-lg transition-all"
          >
            {isTax ? 'Start Free Tax Simulation 👉' : 'Explore K-Market Now 👉'}
          </Link>
        </div>
      </article>
    </div>
  );
}
```
